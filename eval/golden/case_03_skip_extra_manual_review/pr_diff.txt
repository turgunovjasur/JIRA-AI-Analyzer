+create or replace package body Stock_Export is
+
+  function Export_Xlsx(p_filter in varchar2) return blob is
+    v_book Xlsx_Builder.t_book;
+  begin
+    Xlsx_Builder.Add_Header(v_book,
+      t_columns('Mahsulot kodi', 'Nomi', 'Ombor', 'Qoldiq', 'O''lchov'));
+    for r in (select * from v_stock_balance where dyn_filter(p_filter) = 1) loop
+      Xlsx_Builder.Add_Row(v_book,
+        t_values(r.product_code, r.product_name, r.warehouse, r.quantity, r.uom));
+    end loop;
+    return Xlsx_Builder.To_Blob(v_book);
+  end;
+
+end Stock_Export;
