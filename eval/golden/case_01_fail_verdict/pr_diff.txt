+create or replace package body Sms_Alert is
+
+  procedure Send_Debt_Sms is
+    v_template varchar2(4000);
+  begin
+    for r_client in (
+      select c.id, c.name, c.balance
+        from clients c
+       where c.balance < -50000
+    ) loop
+      select t.body
+        into v_template
+        from sms_templates t
+       where t.code = 'DEBT_ALERT';
+
+      Sms_Gateway.Send(
+        p_client_id => r_client.id,
+        p_text      => replace(v_template, '{name}', r_client.name)
+      );
+    end loop;
+  end;
+
+end Sms_Alert;
